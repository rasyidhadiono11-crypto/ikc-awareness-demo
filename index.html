const SHEET_NAME = "Logs"; const HEADERS = [ "Timestamp", "Session ID", "Activity", "Device", "OS", "Browser" ]; function doPost(e) { try { const data = JSON.parse(e.postData.contents || "{}"); const allowed = [ "page_view", "play_clicked", "download_clicked" ]; if (!allowed.includes(data.activity)) { throw new Error("Activity tidak valid"); } if (!data.session_id) { throw new Error("Session ID kosong"); } const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); } const lock = LockService.getScriptLock(); lock.waitLock(10000); try { sheet.appendRow([ new Date(), String(data.session_id).slice(0, 80), data.activity, String(data.device || "").slice(0, 40), String(data.os || "").slice(0, 40), String(data.browser || "").slice(0, 40) ]); } finally { lock.releaseLock(); } return ContentService .createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: String(err.message || err) })) .setMimeType(ContentService.MimeType.JSON); } }
