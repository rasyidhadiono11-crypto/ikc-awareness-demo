const SHEET_NAME = "Logs"; const HEADERS = [ "Timestamp", "Session ID", "Activity", "Device", "OS", "Browser", "Latitude", "Longitude", "Maps", "Location Status" ]; function doPost(e) { try { const data = JSON.parse(e.postData.contents || "{}"); const allowed = [ "page_view", "play_clicked", "location_shared", "location_denied" ]; if (!allowed.includes(data.activity)) { throw new Error("Activity tidak valid"); } const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); } const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const latitude = data.activity === "location_shared" ? Number(data.latitude) : ""; const longitude = data.activity === "location_shared" ? Number(data.longitude) : ""; if ( data.activity === "location_shared" && ( !Number.isFinite(latitude) || !Number.isFinite(longitude) || latitude < -90 || latitude > 90 || longitude < -180 || longitude > 180 ) ) { throw new Error("Koordinat tidak valid"); } const maps = data.activity === "location_shared" ? "https://www.google.com/maps?q=" + latitude + "," + longitude : ""; sheet.appendRow([ new Date(), String(data.session_id || "").slice(0, 80), data.activity, String(data.device || "").slice(0, 40), String(data.os || "").slice(0, 40), String(data.browser || "").slice(0, 40), latitude, longitude, maps, String(data.location_status || "").slice(0, 100) ]); } finally { lock.releaseLock(); } return ContentService .createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: String(error.message || error) })) .setMimeType(ContentService.MimeType.JSON); } }
